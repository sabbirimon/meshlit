cmake_minimum_required(VERSION 3.22.1)
project(meshlit_hyperl LANGUAGES C)
add_library(meshlit_hyperl_cpu STATIC cpu.c)
target_include_directories(meshlit_hyperl_cpu PUBLIC include)
set_target_properties(meshlit_hyperl_cpu PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(meshlit_hyperl_cpu PRIVATE /W4 /fp:strict)
else()
    target_compile_options(meshlit_hyperl_cpu PRIVATE -Wall -Wextra -Werror -ffp-contract=off -fvisibility=hidden)
endif()
if(ANDROID)
    add_library(meshlit_hyperl SHARED android_jni.c)
    target_link_libraries(meshlit_hyperl PRIVATE meshlit_hyperl_cpu)
    target_compile_options(meshlit_hyperl PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
    target_link_options(meshlit_hyperl PRIVATE -Wl,-z,max-page-size=16384)
else()
    enable_testing()
    foreach(check IN ITEMS cpu_test cpu_contract_stress)
        add_executable(${check} ${check}.c)
        target_link_libraries(${check} PRIVATE meshlit_hyperl_cpu)
        if(MSVC)
            target_compile_options(${check} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${check} PRIVATE -UNDEBUG)
            target_link_libraries(${check} PRIVATE m)
        endif()
        add_test(NAME ${check} COMMAND ${check})
    endforeach()
endif()
