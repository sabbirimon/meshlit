cmake_minimum_required(VERSION 3.22.1)
project(meshlit_gibberlink LANGUAGES CXX)
add_library(meshlit_gibberlink SHARED bridge.cpp third_party/ggwave/src/ggwave.cpp)
target_include_directories(meshlit_gibberlink PRIVATE third_party/ggwave/include third_party/ggwave/src)
set_target_properties(meshlit_gibberlink PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(meshlit_gibberlink PRIVATE -Wall -Wextra -fvisibility=hidden)
if(ANDROID)
    target_link_options(meshlit_gibberlink PRIVATE -Wl,-z,max-page-size=16384)
else()
    find_package(JNI REQUIRED)
    target_include_directories(meshlit_gibberlink PRIVATE ${JNI_INCLUDE_DIRS})
endif()
