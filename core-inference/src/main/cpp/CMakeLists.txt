cmake_minimum_required(VERSION 3.22)
project(meshlit_placement LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(meshlit_layer_kernel STATIC placement.c)
set_target_properties(meshlit_layer_kernel PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(MESHLIT_BUILD_JNI "Build the Android compatibility bridge for host tests" OFF)
if(ANDROID OR MESHLIT_BUILD_JNI)
    add_library(meshlit_placement SHARED placement_jni.c)
    target_link_libraries(meshlit_placement PRIVATE meshlit_layer_kernel)
    if(ANDROID)
        target_link_options(meshlit_placement PRIVATE -Wl,-z,max-page-size=16384)
    else()
        find_package(JNI REQUIRED)
        target_include_directories(meshlit_placement PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()
if(NOT ANDROID)
    enable_testing()
    add_executable(check_layer_placement placement_check.c)
    target_link_libraries(check_layer_placement PRIVATE meshlit_layer_kernel)
    add_test(NAME layer_placement COMMAND check_layer_placement)
endif()
